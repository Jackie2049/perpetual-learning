import ast
import asyncio
import __future__
from pathlib import Path
import subprocess
from types import ModuleType
from typing import Any, Optional

import numpy as np
from pydantic import BaseModel
import pytest

REPO = Path(__file__).parent / "verl-7913"
BASE = "3c6d9b43b67c74ec343d77874cd8fa6e841523da"
HEAD = "cd99daf82e4ce52afd1fe02eb47956f0cb381626"

def parsed(revision, path):
    text = subprocess.check_output(["git", "show", f"{revision}:{path}"], cwd=REPO, text=True)
    compile(text, path, "exec")  # Full-file syntax only.
    return ast.parse(text)


def execute(nodes, namespace):
    tree = ast.fix_missing_locations(ast.Module(body=nodes, type_ignores=[]))
    exec(compile(tree, "original-git-objects", "exec", flags=__future__.annotations.compiler_flag), namespace)


for revision in (BASE, HEAD):

    scope = module.__dict__
    scope.update(asyncio=asyncio, np=np, BaseModel=BaseModel, Any=Any, Optional=Optional)
    for path, names in (
        ("verl/workers/rollout/replica.py", {"TokenOutput"}),
        ("verl/utils/tokenizer/tokenizer.py", {"normalize_token_ids"}),
        ("verl/workers/rollout/llm_server.py", {"LLMServerClient", "FullyAsyncLLMServerClient"}),
    ):
        nodes = [node for node in parsed(revision, path).body if getattr(node, "name", None) in names]
        for node in nodes:
            if isinstance(node, ast.ClassDef):
                for member in node.body:
                    if isinstance(member, (ast.FunctionDef, ast.AsyncFunctionDef)):

    tests = {"llm_server": module, "FullyAsyncLLMServerClient": scope["FullyAsyncLLMServerClient"],

    cases = [

        ("test_resume_past_min_tokens_floors_at_zero", ()),
        ("test_without_min_tokens_adds_none", ()),
    ]
    passed = 0
    for name, args in cases:
