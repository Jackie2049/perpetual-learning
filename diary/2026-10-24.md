from types import ModuleType
from unittest.mock import patch

REPO = Path(__file__).parent / "unirl-readonly"

             "unirl.models.minimax_h3", "unirl.models.minimax_h3.vendor"):

    sys.modules[name] = module

node = next(n for n in tree.body if isinstance(n, ast.Assign)
            and any(isinstance(t, ast.Name) and t.id == "MINIMAX_H3_TEXT_ENCODER_LAYER" for t in n.targets))
exec(compile(ast.Module(body=[node], type_ignores=[]), "original-layer-constant", "exec"),
     sys.modules["unirl.models.minimax_h3.vendor"].__dict__)


def load(name, path):
    spec = importlib.util.spec_from_file_location(name, REPO / path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module


load("unirl.config.require", "unirl/config/require.py")
load("unirl.utils.dtypes", "unirl/utils/dtypes.py")
store = load("unirl.models.minimax_h3.offline_text_embed", "unirl/models/minimax_h3/offline_text_embed.py")


def writer(path, **extra):
    options = dict(model_checkpoint="fixture-model-A", dtype="bfloat16", shard_size=2,
                   resume=False, force_overwrite=False)
    options.update(extra)
    return store.OfflineTextEmbedWriter(str(path), **options)


def rejects(call):
    try:
        call()
    except ValueError as error:
        return str(error)
    raise AssertionError("Expected ValueError")


result = {"torch": torch.__version__, "safetensors": safetensors.__version__}
with tempfile.TemporaryDirectory(prefix="unirl-offline-store-") as temp:
    root = Path(temp)
    cache = root / "cache"
    tensors = {"prompt": torch.ones((1, 5120), dtype=torch.bfloat16),
               " prompt ": torch.full((2, 5120), 2, dtype=torch.bfloat16),
               "third": torch.full((3, 5120), 3, dtype=torch.bfloat16)}
    out = writer(cache)
    for prompt, tensor in tensors.items():
        out.add(prompt, tensor)
    out.close()
    assert len({store.compute_prompt_key(p) for p in tensors}) == 3
    reader = store.OfflineTextEmbedStore.from_dir(str(cache))
    for prompt, tensor in tensors.items():
        assert torch.equal(reader.get(prompt), tensor)
    reader.verify_coverage_or_raise(list(tensors), context="fixture")
    result["round_trip"] = {"prompts": 3, "shards": 2, "exact_whitespace_keys": True}
    for changed in (dict(model_checkpoint="fixture-model-B"), dict(dtype="float32")):

    result["resume_metadata_rejections"] = 2
    index_path = cache / "index.json"
    original = index_path.read_bytes()
    index = json.loads(original)
    altered = {**index, "model_checkpoint": "fixture-model-B", "dtype": "float32"}
    index_path.write_text(json.dumps(altered))
    assert store.OfflineTextEmbedStore.from_dir(str(cache)).get("prompt").dtype == torch.bfloat16
    index_path.write_bytes(original)
    result["reader_metadata_is_not_runtime_identity_check"] = True
    broken = root / "missing-shards"
    broken.mkdir()
    (broken / "index.json").write_bytes(original)
    broken_reader = store.OfflineTextEmbedStore.from_dir(str(broken))
    broken_reader.verify_coverage_or_raise(list(tensors), context="fixture-missing-shards")
    rejects(lambda: broken_reader.get("prompt"))
    result["coverage_passes_before_missing_shard_get_fails"] = True
    resumed = writer(cache, resume=True, shard_size=1)
    with patch.object(store.os, "replace", side_effect=OSError("injected publication failure")):
        try:
            resumed.add("fourth", torch.ones((1, 5120), dtype=torch.bfloat16))
        except OSError as error:
            assert str(error) == "injected publication failure"
        else:
            raise AssertionError("Expected injected publication failure")
    assert index_path.read_bytes() == original
    assert (cache / "embeddings_00002.safetensors").exists()
    assert (cache / "index.json.tmp").exists()
    retry = writer(cache, resume=True, shard_size=1)
    assert store.compute_prompt_key("fourth") not in retry.entries and len(retry.entries) == 3
    retry.add("fourth", torch.full((1, 5120), 4, dtype=torch.bfloat16))
    retry.close()
    fresh = store.OfflineTextEmbedStore.from_dir(str(cache))
    for prompt, tensor in tensors.items():

"""Original judge method with real requests.Response fixtures; no HTTP traffic."""
import ast
import json
import logging
from pathlib import Path
import re
from types import SimpleNamespace
from unittest.mock import patch

import requests

REPO = Path(__file__).parent / "unirl-readonly"
source = (REPO / "unirl/reward/local/llm_judge.py").read_text()
compile(source, "llm_judge.py", "exec")
tree = ast.parse(source)
selected = [node for node in tree.body if isinstance(node, ast.Assign)
            and any(isinstance(target, ast.Name) and target.id in
                    {"_DEFAULT_JUDGE_PROMPT", "_INCORRECT_RE", "_CORRECT_RE"}
                    for target in node.targets)]
cls = next(node for node in tree.body if isinstance(node, ast.ClassDef)
           and node.name == "LLMJudgeRewardScorer")
method = next(node for node in cls.body if isinstance(node, ast.FunctionDef) and node.name == "_judge_one")
scope = {"re": re, "requests": requests, "logger": logging.getLogger("judge-protocol-control")}
exec(compile(ast.fix_missing_locations(ast.Module(body=selected + [method], type_ignores=[])),
             "original-judge-method", "exec"), scope)
scope["logger"].disabled = True
spec = SimpleNamespace(prompt_template=scope["_DEFAULT_JUDGE_PROMPT"], max_prediction_chars=4000,
                       max_tokens=512, max_retries=3, timeout=0.1)
scorer = SimpleNamespace(_spec=spec, _endpoint="blocked-fixture-only", _judge_model="fixture", _api_key="")
cases = [
    ("positive", 200, {"choices": [{"message": {"content": "correct"}}]}, 1.0, 1),
    ("negative", 200, {"choices": [{"message": {"content": "incorrect"}}]}, 0.0, 1),
    ("negation_not_adjacent", 200,
     {"choices": [{"message": {"content": "This is not the correct answer."}}]}, 1.0, 1),
    ("unrecognized", 200, {"choices": [{"message": {"content": "I cannot judge."}}]}, 0.0, 1),
    ("earlier_negative_later_positive", 200,
     {"choices": [{"message": {"content": "At first I thought incorrect. Final verdict: correct."}}]}, 0.0, 1),
    ("empty_choices", 200, {"choices": []}, 0.0, 3),
    ("http_503", 503, {"error": "fixture-unavailable"}, 0.0, 3),
]
results = []
for name, status, payload, expected, expected_calls in cases:
    response = requests.Response()
    response.status_code = status
    response._content = json.dumps(payload).encode()
    response.url = "blocked-fixture-only"
    with patch.object(requests, "post", return_value=response) as post:
        score = scope["_judge_one"](scorer, "question", "prediction", "answer")
        assert score == expected and post.call_count == expected_calls
        for call in post.call_args_list:
            assert call.args == ("blocked-fixture-only",)
            assert call.kwargs["timeout"] == 0.1
            assert "Authorization" not in call.kwargs["headers"]
        results.append({"case": name, "score": score, "transport_stub_calls": post.call_count})
print(json.dumps({"requests": requests.__version__, "original_method_cases": results}, indent=2))

"""Isolate the exact upstream guard; fixtures are not a TrainStack run."""
import ast
import dataclasses
import hashlib
import subprocess
import sys
from types import SimpleNamespace as NS

REPO = sys.argv[1]
PATH = "unirl/train/stack/single.py"
REVS = ("784893d10c01ab21fdf2e33fbd7cec18123bf80f", "471690694c51f3070e012276429585fcae877a4e")

@dataclasses.dataclass
class Result:
    has_backward: bool
    micros: list
    metrics: dict

def micro(count, metrics, backward=True):
    return NS(num_steps_or_tokens=count, metrics=metrics, has_backward=backward)

for rev in REVS:
    source = subprocess.check_output(["git", "-C", REPO, "show", f"{rev}:{PATH}"])
    tree = ast.parse(source)
    cls = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "TrainStack")
    method = next(n for n in cls.body if isinstance(n, ast.FunctionDef) and n.name == "_guard_first_update_ratio")
    module = ast.Module(body=[ast.ImportFrom(module="__future__", names=[ast.alias(name="annotations")], level=0), method], type_ignores=[])
    warnings = []
    scope = {"replace": dataclasses.replace, "logger": NS(warning=lambda *args: warnings.append(args))}
    exec(compile(ast.fix_missing_locations(module), PATH, "exec"), scope)
    guard = scope[method.name]
    cases = [
        ("empty", [micro(0, {})], True, 0.1, 0.0, 0),
        ("mixed", [micro(0, {}), micro(4, {"ratio_min": 0.8, "ratio_max": 1.1})], True, 0.1, 0.2, 1),
        ("valid", [micro(4, {"ratio_min": 0.8, "ratio_max": 1.1})], True, 0.1, 0.2, 1),
        ("disabled", [micro(0, {})], True, None, None, 0),
        ("no_backward", [micro(0, {}, False)], False, 0.1, None, 0),
        ("ignored_non_backward", [micro(0, {}, False), micro(2, {"ratio_min": 1.0, "ratio_max": 1.0})], True, 0.1, 0.0, 0),
    ]
    print(rev, "source_sha256", hashlib.sha256(source).hexdigest())
    for name, micros, backward, tolerance, expected, warning_count in cases:
        warnings.clear()
        result = Result(backward, micros, {"existing": 7})
        try:
            out = guard(NS(first_update_ratio_tol=tolerance), result)
        except KeyError as exc:
            assert rev == REVS[0] and name in ("empty", "mixed") and exc.args == ("ratio_min",)
            print(name, "KeyError:ratio_min")
            continue
        assert out.metrics["existing"] == 7
        assert result.metrics == {"existing": 7}
        if expected is None:
            assert out is result
        else:
            assert abs(out.metrics["first_update_ratio_dev"] - expected) < 1e-12
        assert len(warnings) == warning_count
        print(name, "PASS", out.metrics, "warnings", len(warnings))

import hashlib
import multiprocessing as mp
import os
import subprocess
import sys

HEAD = "6fd82cba13f52f394747832344039408699ea378"

def source(repo, path):
    return subprocess.check_output(["git", "-C", repo, "show", f"{HEAD}:{path}"],
                                   env=dict(os.environ, GIT_NO_LAZY_FETCH="1"))

def worker(conn, repo, previous, count):
    namespace = {"__name__": "original_runtime_helpers"}
    exec(compile(source(repo, "unirl/rollout/engine/vllm/runtime.py"), "runtime.py", "exec"), namespace)
    try:
        for _ in range(count):
            try:
                request = namespace["recv_request"](conn, last_request_id=previous)
            except namespace["ProtocolError"] as error:
                # Classification reply belongs to this harness, not runtime error policy.
                conn.send({"caught": type(error).__name__, "message": str(error)})
                return
            previous = request["request_id"]
            namespace["send_response"](conn, request, ok=True, result={"echo": request})
    finally:
        conn.close()

def exchange(repo, messages, previous=0):
    ctx = mp.get_context("spawn")
    parent, child = ctx.Pipe(duplex=True)
    process = ctx.Process(target=worker, args=(child, repo, previous, len(messages)))
    process.start()
    child.close()
    responses = []
    try:
        for message in messages:
            parent.send(message)
            assert parent.poll(5), "helper child did not reply"
            responses.append(parent.recv())
        process.join(5)
        assert process.exitcode == 0, process.exitcode
    finally:
        parent.close()
        if process.is_alive():
            process.terminate()
            process.join(5)
    return responses

def main(repo):
